const RECIPES_KEY = 'kitchenVorrat_recipes';
const WEEKPLAN_KEY = 'kitchenVorrat_weekplan';
const STORES_KEY = 'kitchenVorrat_stores';
const PRICE_HISTORY_KEY = 'kitchenVorrat_price_history';

// ═══════════════════════════════════════════════════════════
//  REZEPTE — DATA LAYER
// ═══════════════════════════════════════════════════════════
function loadRecipes() {
  try {
    const data = localStorage.getItem(RECIPES_KEY);
    return data ? JSON.parse(data) : [];
  } catch (e) {
    console.error('loadRecipes:', e);
    return [];
  }
}

function saveRecipes(recipes) {
  try {
    localStorage.setItem(RECIPES_KEY, JSON.stringify(recipes));
  } catch (e) {
    console.error('saveRecipes:', e);
  }
}

function addRecipe(recipe) {
  const recipes = loadRecipes();
  recipe.id = 'r-' + Date.now();
  recipe.created = new Date().toISOString();
  recipe.ingredients = recipe.ingredients || [];
  recipe.steps = recipe.steps || [];
  recipes.push(recipe);
  saveRecipes(recipes);
  renderRecipeList();
  return recipe;
}

function updateRecipe(id, updates) {
  const recipes = loadRecipes();
  const idx = recipes.findIndex(r => r.id === id);
  if (idx >= 0) {
    Object.assign(recipes[idx], updates);
    saveRecipes(recipes);
    renderRecipeList();
  }
}

function deleteRecipe(id) {
  if (!confirm('Rezept wirklich löschen?')) return;
  const recipes = loadRecipes().filter(r => r.id !== id);
  saveRecipes(recipes);
  closeRecipeDetail();
  renderRecipeList();
}

function getRecipeById(id) {
  return loadRecipes().find(r => r.id === id);
}

function searchRecipes(query) {
  const recipes = loadRecipes();
  if (!query) return recipes;
  const q = query.toLowerCase();
  return recipes.filter(r =>
    r.name.toLowerCase().includes(q) ||
    (r.cuisine || '').toLowerCase().includes(q) ||
    (r.ingredients || []).some(i => i.name.toLowerCase().includes(q))
  );
}

// ═══════════════════════════════════════════════════════════
//  REZEPTE — RENDERING
// ═══════════════════════════════════════════════════════════
function renderRecipeList() {
  const container = document.getElementById('recipeList');
  const search = document.getElementById('recipeSearch')?.value || '';
  const recipes = searchRecipes(search);

  if (recipes.length === 0) {
    container.innerHTML = `
      <div style="text-align:center;padding:40px 20px;color:var(--text-tertiary);">
        <div style="font-size:48px;margin-bottom:12px;">🍳</div>
        <p>${search ? 'Keine Rezepte gefunden' : 'Noch keine Rezepte'}</p>
        ${!search ? '<button class="btn btn-primary" style="margin-top:12px;" onclick="openRecipeModal()"><i class="fas fa-plus"></i> Neues Rezept</button>' : ''}
      </div>`;
    return;
  }

  container.innerHTML = recipes.map(r => {
    const ingPreview = (r.ingredients || []).slice(0, 4).map(i => i.name).join(', ');
    const moreText = (r.ingredients || []).length > 4 ? ` +${(r.ingredients || []).length - 4}` : '';
    return `
      <div class="recipe-card">
        <div class="rc-header" onclick="openRecipeDetail('${r.id}')">
          <span class="rc-emoji">${r.emoji || '🍽️'}</span>
          <div>
            <div class="rc-name">${r.name}</div>
            <div class="rc-meta">
              <span>⏱ ${r.time || '?'} Min</span>
              <span>👥 ${r.servings || '?'} Port.</span>
            </div>
          </div>
        </div>
        ${ingPreview ? `<div class="rc-ingredients-preview">${ingPreview}${moreText}</div>` : ''}
        <div class="rc-actions">
          <button class="btn btn-primary" onclick="openRecipeDetail('${r.id}')"><i class="fas fa-eye"></i> Ansehen</button>
          <button class="btn btn-secondary" onclick="addRecipeToWeekplan('${r.id}')"><i class="fas fa-calendar-plus"></i> Plan</button>
          <button class="btn btn-secondary" onclick="openRecipeModal('${r.id}')"><i class="fas fa-pen"></i></button>
          <button class="btn btn-secondary" onclick="deleteRecipe('${r.id}')"><i class="fas fa-trash"></i></button>
        </div>
      </div>`;
  }).join('');
}

      <div>
        <div class="rd-name">${recipe.name}</div>
        <div style="font-size:13px;color:var(--text-secondary);">${recipe.cuisine || ''} · ${recipe.difficulty || 'mittel'}</div>
      </div>
    </div>
    <div class="rd-meta">
      <span>⏱ ${recipe.time || '-'} Min</span>
      <span>👥 ${recipe.servings || '-'} Port.</span>
      ${recipe.calories ? '<span>🔥 ' + recipe.calories + ' kcal</span>' : ''}
    </div>

    <div>${(recipe.ingredients || []).map((ing, i) => {
      const status = missingIngredients.status[i] || 'missing';
      return `
        <div class="ri-row">
          <div class="ri-check ${status === 'available' ? 'checked' : ''}" onclick="toggleIngredientCheck('${recipe.id}', ${i})"></div>

          <span class="ri-qty">${ing.qty || ''} ${ing.unit || ''}</span>

        </div>`;
    }).join('')}</div>

    <div style="display:flex;gap:8px;margin-top:12px;">
      <button class="btn btn-primary" style="flex:1;" onclick="addMissingToShopping('${recipe.id}')">
        <i class="fas fa-cart-plus"></i> Fehlende einkaufen
      </button>
      <button class="btn btn-secondary" style="flex:1;" onclick="addRecipeToWeekplan('${recipe.id}')">
        <i class="fas fa-calendar-plus"></i> In Wochenplan
      </button>

    ${recipe.steps && recipe.steps.length > 0 ? `

}

function closeRecipeDetail() {
  document.getElementById('recipeDetailBackdrop').style.display = 'none';
}

function toggleIngredientCheck(recipeId, ingIdx) {
  // Temporärer Check-State im localStorage
  const detailKey = `rd_${recipeId}`;
  let checked = JSON.parse(localStorage.getItem(detailKey) || '{}');
  checked[ingIdx] = !checked[ingIdx];
  localStorage.setItem(detailKey, JSON.stringify(checked));
  openRecipeDetail(recipeId);
}

// ═══════════════════════════════════════════════════════════
//  ZUTATEN-ABGLEICH mit Vorrat
// ═══════════════════════════════════════════════════════════
function checkRecipeIngredients(recipe) {
  const products = loadProducts();
  const status = [];
  let missingCount = 0;

  (recipe.ingredients || []).forEach(ing => {
    const match = products.find(p =>
      p.name.toLowerCase().includes(ing.name.toLowerCase()) ||
      ing.name.toLowerCase().includes(p.name.toLowerCase())
    );
    if (match && match.qty >= 1) {
      status.push('available');
    } else {
      status.push('missing');
      missingCount++;
    }
  });

  return { status, missing: missingCount };
}

function getMissingIngredients(recipe) {
  const products = loadProducts();
  const result = [];
  (recipe.ingredients || []).forEach(ing => {
    const match = products.find(p =>
      p.name.toLowerCase().includes(ing.name.toLowerCase()) ||
      ing.name.toLowerCase().includes(p.name.toLowerCase())
    );
    if (!match || match.qty < (parseFloat(ing.qty) || 0)) {
      result.push({ ...ing, has: match ? match.qty : 0 });
    }
  });
  return result;
}

function addMissingToShopping(recipeId) {
  const recipe = getRecipeById(recipeId);
  if (!recipe) return;
  const missing = getMissingIngredients(recipe);
  missing.forEach(ing => {
    addToShopping(ing.name, ing.qty, ing.unit, ing.category || getCategoryForIngredient(ing.name));
  });
  showToast(`${missing.length} fehlende Zutaten zur Einkaufsliste hinzugefügt`);
  saveShopping(loadShopping());
  closeRecipeDetail();
}

function getCategoryForIngredient(name) {
  const n = name.toLowerCase();
  if (n.includes('milch') || n.includes('käse') || n.includes('butter') || n.includes('joghurt')) return 'Milchprodukte';
  if (n.includes('fleisch') || n.includes('hähnchen') || n.includes('lachs') || n.includes('fisch')) return 'Fleisch & Fisch';
  if (n.includes('gemüse') || n.includes('tomate') || n.includes('zwiebel') || n.includes('kartoffel')) return 'Gemüse';
  if (n.includes('obst')) return 'Obst';
  if (n.includes('nudel') || n.includes('reis') || n.includes('mehl') || n.includes('brot')) return 'Getreideprodukte';
  return 'Sonstige';
}

let editingRecipeId = null;

function openRecipeModal(id) {
  editingRecipeId = id || null;
  document.getElementById('recipeModalTitle').textContent = id ? 'Rezept bearbeiten' : 'Neues Rezept';

  if (id) {
    const r = getRecipeById(id);
    document.getElementById('recipeName').value = r.name || '';
    document.getElementById('recipeEmoji').value = r.emoji || '';
    document.getElementById('recipeCuisine').value = r.cuisine || 'italienisch';
    document.getElementById('recipeDifficulty').value = r.difficulty || 'mittel';
    document.getElementById('recipeTime').value = r.time || '';
    document.getElementById('recipeServings').value = r.servings || '';
    document.getElementById('recipeCalories').value = r.calories || '';
    document.getElementById('recipeNotes').value = r.notes || '';
    renderRecipeModalIngredients(r.ingredients || []);
    renderRecipeModalSteps(r.steps || []);
  } else {
    document.getElementById('recipeName').value = '';
    document.getElementById('recipeEmoji').value = '';
    document.getElementById('recipeCuisine').value = 'italienisch';
    document.getElementById('recipeDifficulty').value = 'mittel';
    document.getElementById('recipeTime').value = '';
    document.getElementById('recipeServings').value = '';
    document.getElementById('recipeCalories').value = '';
    document.getElementById('recipeNotes').value = '';
    renderRecipeModalIngredients([]);
    renderRecipeModalSteps([]);
  }

  document.getElementById('recipeModalBackdrop').style.display = 'flex';
}

function closeRecipeModal() {
  document.getElementById('recipeModalBackdrop').style.display = 'none';
  editingRecipeId = null;
}

function renderRecipeModalIngredients(ingredients) {
  const container = document.getElementById('recipeIngredients');
  container.innerHTML = ingredients.map((ing, i) => `
    <div class="ri-row" style="background:var(--bg);border-radius:var(--radius-sm);padding:8px;margin-bottom:4px;">
      <input type="text" class="form-input" placeholder="Name" value="${ing.name || ''}" style="flex:2;font-size:14px;" data-ing-name="${i}">
      <input type="text" class="form-input" placeholder="Menge" value="${ing.qty || ''}" style="flex:0.6;font-size:14px;" data-ing-qty="${i}">

function renderRecipeModalSteps(steps) {
  const container = document.getElementById('recipeSteps');
  container.innerHTML = steps.map((s, i) => `
    <div style="display:flex;gap:8px;margin-bottom:8px;align-items:start;">
      <span style="width:24px;height:24px;border-radius:50%;background:var(--bg);display:flex;align-items:center;justify-content:center;font-size:12px;font-weight:700;flex-shrink:0;">${i + 1}</span>

function addRecipeIngredient() {
  const inputs = document.querySelectorAll('[data-ing-name]');
  const ingredients = [];
  inputs.forEach(inp => {
    const idx = parseInt(inp.dataset.ingName);
    ingredients[idx] = {
      name: inp.value,
      qty: document.querySelector(`[data-ing-qty="${idx}"]`)?.value || '',
      unit: document.querySelector(`[data-ing-unit="${idx}"]`)?.value || ''
    };
  });
  ingredients.push({ name: '', qty: '', unit: '' });
  renderRecipeModalIngredients(ingredients);
}

function removeRecipeIngredient(idx) {
  const inputs = document.querySelectorAll('[data-ing-name]');
  const ingredients = [];
  inputs.forEach(inp => {
    const i = parseInt(inp.dataset.ingName);
    if (i !== idx) {
      ingredients[i] = {
        name: inp.value,
        qty: document.querySelector(`[data-ing-qty="${i}"]`)?.value || '',
        unit: document.querySelector(`[data-ing-unit="${i}"]`)?.value || ''
      };
    }
  });
  renderRecipeModalIngredients(ingredients);
}

function addRecipeStep() {
  const inputs = document.querySelectorAll('[data-step]');
  const steps = [];
  inputs.forEach(inp => {
    steps.push(inp.value);
  });
  steps.push('');
  renderRecipeModalSteps(steps);
}

function removeRecipeStep(idx) {
  const inputs = document.querySelectorAll('[data-step]');
  const steps = [];
  inputs.forEach(inp => {
    const i = parseInt(inp.dataset.step);
    if (i !== idx) steps.push(inp.value);
  });
  renderRecipeModalSteps(steps);
}

function saveRecipeModal() {
  const name = document.getElementById('recipeName').value.trim();
  if (!name) { showToast('Bitte Namen eingeben'); return; }

  // Zutaten sammeln
  const ingInputs = document.querySelectorAll('[data-ing-name]');
  const ingredients = [];
  ingInputs.forEach(inp => {
    const idx = parseInt(inp.dataset.ingName);
    const ing = {
      name: inp.value.trim(),
      qty: document.querySelector(`[data-ing-qty="${idx}"]`)?.value || '',
      unit: document.querySelector(`[data-ing-unit="${idx}"]`)?.value || ''
    };
    if (ing.name) ingredients.push(ing);
  });

  // Schritte sammeln
  const stepInputs = document.querySelectorAll('[data-step]');
  const steps = [];
  stepInputs.forEach(inp => {
    const s = inp.value.trim();
    if (s) steps.push(s);
  });

  const recipeData = {
    name,
    emoji: document.getElementById('recipeEmoji').value.trim() || '🍽️',
    cuisine: document.getElementById('recipeCuisine').value,
    difficulty: document.getElementById('recipeDifficulty').value,
    time: parseInt(document.getElementById('recipeTime').value) || 0,
    servings: parseInt(document.getElementById('recipeServings').value) || 4,
    calories: parseInt(document.getElementById('recipeCalories').value) || 0,
    notes: document.getElementById('recipeNotes').value.trim(),
    ingredients,
    steps
  };

  if (editingRecipeId) {
    updateRecipe(editingRecipeId, recipeData);
    showToast('Rezept aktualisiert ✓');
  } else {
    addRecipe(recipeData);
    showToast('Rezept erstellt ✓');
  }
  closeRecipeModal();
}

function loadWeekPlan() {
  try {
    const data = localStorage.getItem(WEEKPLAN_KEY);
    return data ? JSON.parse(data) : {};
  } catch (e) {
    return {};
  }
}

function saveWeekPlan(plan) {
  localStorage.setItem(WEEKPLAN_KEY, JSON.stringify(plan));
}

function getWeekDays() {
  const today = new Date();
  const dayOfWeek = today.getDay(); // 0=Sun, 1=Mon...
  const monday = new Date(today);
  monday.setDate(today.getDate() - ((dayOfWeek + 6) % 7)); // This Monday

  const days = [];
  const dayNames = ['Mo', 'Di', 'Mi', 'Do', 'Fr', 'Sa', 'So'];
  for (let i = 0; i < 7; i++) {
    const d = new Date(monday);
    d.setDate(monday.getDate() + i);
    days.push({
      key: d.toISOString().split('T')[0],
      name: dayNames[i],
      num: d.getDate(),
      isToday: d.toDateString() === today.toDateString()
    });
  }
  return days;
}

// ═══════════════════════════════════════════════════════════
//  WOCHENPLAN — RENDERING
// ═══════════════════════════════════════════════════════════
let selectedDayKey = null;

function renderWeekPlan() {
  const plan = loadWeekPlan();
  const days = getWeekDays();
  const grid = document.getElementById('weekPlanGrid');

  if (!selectedDayKey) {
    selectedDayKey = days.find(d => d.isToday)?.key || days[0].key;
  }

  grid.innerHTML = days.map(d => {
    const meals = plan[d.key] || [];
    const mealPreview = meals.length > 0 ? meals[0].name.substring(0, 12) : '';
    return `
      <div class="week-day ${d.isToday ? 'today' : ''} ${d.key === selectedDayKey ? 'selected' : ''}" onclick="selectDay('${d.key}')">
        <div class="wd-name">${d.name}</div>
        <div class="wd-num">${d.num}</div>
        <div class="wd-meal">${mealPreview}</div>
      </div>`;
  }).join('');

  renderDayDetail();
}

function selectDay(key) {
  selectedDayKey = key;
  renderWeekPlan();
}

function renderDayDetail() {
  const plan = loadWeekPlan();
  const days = getWeekDays();
  const day = days.find(d => d.key === selectedDayKey);
  const meals = plan[selectedDayKey] || [];
  const container = document.getElementById('dayDetail');

  const mealTypeEmojis = { breakfast: '🌅', lunch: '☀️', dinner: '🌙', snack: '🍪' };
  const mealTypeNames = { breakfast: 'Frühstück', lunch: 'Mittagessen', dinner: 'Abendessen', snack: 'Snack' };

  container.innerHTML = `
    <div style="display:flex;justify-content:space-between;align-items:center;margin-bottom:8px;">
      <span style="font-size:15px;font-weight:600;">${day?.name || ''}, ${day?.num || ''}.</span>
    </div>
    ${Object.entries(mealTypeNames).map(([type, label]) => {
      const typeMeals = meals.filter(m => m.type === type);
      return `
        <div style="margin-bottom:8px;">
          <div style="font-size:13px;font-weight:600;color:var(--text-secondary);margin-bottom:4px;">
            ${mealTypeEmojis[type]} ${label}
          </div>
          ${typeMeals.length > 0 ? typeMeals.map(m => `
            <div class="day-meal-row">
              <span class="dm-type">${mealTypeEmojis[type]}</span>
              <span class="dm-name">${m.name}</span>
              <span class="dm-time">${m.time || ''} Min</span>
              <button class="dm-add" onclick="removeMealFromDay('${selectedDayKey}','${m.recipeId}','${type}')" title="Entfernen"><i class="fas fa-times"></i></button>
            </div>`).join('') : ''}
          <button class="btn btn-secondary" style="margin-top:4px;padding:6px 12px;font-size:12px;" onclick="addMealToDayPrompt('${selectedDayKey}','${type}')">
            <i class="fas fa-plus"></i> Rezept
          </button>
        </div>`;
    }).join('')}
  `;

  // Zutaten-Check
  renderIngredientCheck();
}

function renderIngredientCheck() {
  const container = document.getElementById('ingredientCheck');
  const plan = loadWeekPlan();
  const meals = plan[selectedDayKey] || [];

  if (meals.length === 0) {
    container.innerHTML = `
      <div style="text-align:center;padding:20px;color:var(--text-tertiary);font-size:14px;">
        Keine Rezepte geplant. Füge Rezepte hinzu, um den Zutaten-Check zu sehen.
      </div>`;
    return;
  }

  // Alle Zutaten sammeln
  const allIngredients = {};
  let totalMissing = 0;
  let totalAvailable = 0;

  meals.forEach(meal => {
    const recipe = getRecipeById(meal.recipeId);
    if (!recipe) return;
    const missing = getMissingIngredients(recipe);
    totalMissing += missing.length;
    const availableCount = (recipe.ingredients || []).length - missing.length;
    totalAvailable += availableCount;
    missing.forEach(ing => {
      if (!allIngredients[ing.name]) {
        allIngredients[ing.name] = { ...ing, recipes: [] };
      }
      allIngredients[ing.name].recipes.push(recipe.name);
    });
  });

  const products = loadProducts();
  const entries = Object.values(allIngredients);

  container.innerHTML = entries.length > 0 ? `
    <div style="font-size:13px;color:var(--text-secondary);margin-bottom:8px;">
      ✅ ${totalAvailable} Zutaten da · 🔴 ${totalMissing} fehlen
    </div>
    ${entries.map(ing => {
      const product = products.find(p => p.name.toLowerCase().includes(ing.name.toLowerCase()));
      const has = product && product.qty >= (parseFloat(ing.qty) || 0);
      return `
        <div class="ri-row">
          <span class="ri-status ${has ? 'available' : 'missing'}">${has ? '🟢' : '🔴'}</span>
          <span class="ri-name">${ing.name}</span>
          <span class="ri-qty">${ing.qty} ${ing.unit}</span>
        </div>`;
    }).join('')}` : `
    <div style="text-align:center;padding:20px;color:var(--text-tertiary);font-size:14px;">
      Alle Zutaten vorhanden! 🎉
    </div>`;
}

function addMealToDayPrompt(dayKey, mealType) {
  const recipes = loadRecipes();
  if (recipes.length === 0) {
    showToast('Bitte erst Rezepte erstellen');
    return;
  }

  const listHtml = recipes.map(r => `
    <div style="display:flex;align-items:center;gap:8px;padding:10px 0;border-bottom:0.5px solid var(--separator);cursor:pointer;" onclick="addMealToDay('${dayKey}','${mealType}','${r.id}')">
      <span style="font-size:20px;">${r.emoji || '🍽️'}</span>

function autoWeekPlan() {
  const recipes = loadRecipes();
  if (recipes.length < 7) {
    showToast(`Mindestens 7 Rezepte nötig (hast ${recipes.length})`);
    return;
  }
  const plan = loadWeekPlan();
  const days = getWeekDays();
  const mealTypes = ['lunch', 'dinner'];

  days.forEach((d, i) => {
    if (!plan[d.key]) plan[d.key] = [];
    mealTypes.forEach(mt => {
      const idx = (i * 2 + mealTypes.indexOf(mt)) % recipes.length;
      const r = recipes[idx];
      plan[d.key].push({
        recipeId: r.id, name: r.name, emoji: r.emoji || '🍽️',
        time: r.time || 0, servings: r.servings || 0, type: mt
      });
    });
  });

  saveWeekPlan(plan);
  renderWeekPlan();
  showToast('Wochenplan generiert! 📅');
}

// Haversine formula for distance calculation
function haversineDistance(lat1, lon1, lat2, lon2) {
  const R = 6371; // km
  const dLat = (lat2 - lat1) * Math.PI / 180;
  const dLon = (lon2 - lon1) * Math.PI / 180;
  const a = Math.sin(dLat / 2) * Math.sin(dLat / 2) +
    Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
    Math.sin(dLon / 2) * Math.sin(dLon / 2);
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
  return R * c;
}

function getUserLocation() {
  if (localStorage.getItem('kv_user_lat')) {
    return { lat: parseFloat(localStorage.getItem('kv_user_lat')), lon: parseFloat(localStorage.getItem('kv_user_lon')) };
  }
  // Default: Germany center (Frankfurt area)
  return { lat: 50.1109, lon: 8.6821 };
}

  localStorage.setItem('kv_user_lat', lat.toString());
  localStorage.setItem('kv_user_lon', lon.toString());
}

function renderStoreOfferList() {
  const stores = loadStores();
  const container = document.getElementById('storeOffersList');

  if (stores.length === 0) {
    container.innerHTML = '<div style="text-align:center;padding:20px;color:var(--text-tertiary);">Keine Läden vorhanden</div>';
    return;
  }

  // Alle Angebote sammeln und nach Laden gruppieren
  const allOffers = [];
  stores.forEach(store => {
    (store.offers || []).forEach(offer => {
      allOffers.push({ ...offer, storeName: store.name, storeId: store.id });
    });
  });

  // Nach Datum sortieren (neueste zuerst), dann nach Store
  allOffers.sort((a, b) => new Date(b.date || 0) - new Date(a.date || 0));

  if (allOffers.length === 0) {
    container.innerHTML = '<div style="text-align:center;padding:20px;color:var(--text-tertiary);">Keine Angebote eingetragen</div>';
    return;
  }

  container.innerHTML = allOffers.map(o => {
    const discount = o.originalPrice && o.price ? Math.round((1 - o.price / o.originalPrice) * 100) : 0;
    return `
      <div class="store-offer">
        <span style="font-size:16px;">🏷️</span>
        <div style="flex:1;">
          <div style="font-size:14px;font-weight:500;">${o.productName}</div>
          <div style="font-size:12px;color:var(--text-secondary);">${o.storeName} · ${o.date || ''}</div>

        <span class="so-price">${o.price ? o.price + ' €' : '-'}</span>
        ${o.originalPrice ? `<span class="so-original">${o.originalPrice} €</span>` : ''}
        ${discount > 0 ? `<span class="so-discount">-${discount}%</span>` : ''}
      </div>`;
  }).join('');
}

function loadPriceHistory() {
  try {
    const data = localStorage.getItem(PRICE_HISTORY_KEY);
    return data ? JSON.parse(data) : {};
  } catch (e) { return {}; }
}

function savePriceHistory(history) {
  localStorage.setItem(PRICE_HISTORY_KEY, JSON.stringify(history));
}

function addPriceEntry(productName, price) {
  const history = loadPriceHistory();
  if (!history[productName]) history[productName] = [];
  history[productName].push({
    price: parseFloat(price),
    date: new Date().toISOString().split('T')[0],
    timestamp: Date.now()
  });
  // Sort by date descending
  history[productName].sort((a, b) => b.timestamp - a.timestamp);
  // Keep max 50 entries
  history[productName] = history[productName].slice(0, 50);
  savePriceHistory(history);

  // Also update the product price
  const products = loadProducts();
  const product = products.find(p => p.name === productName);
  if (product) {
    product.price = price.toString();
    saveProducts(products);
  }

  renderPriceTrends();
  renderPriceHistory();
}

function getAvgPrice(productName) {
  const history = loadPriceHistory();
  const entries = history[productName] || [];
  if (entries.length === 0) return null;
  const sum = entries.reduce((a, b) => a + b.price, 0);
  return sum / entries.length;
}

function getPriceTrend(productName) {
  const history = loadPriceHistory();
  const entries = history[productName] || [];
  if (entries.length < 2) return 'stable';
  const firstHalf = entries.slice(0, Math.floor(entries.length / 2));
  const secondHalf = entries.slice(Math.floor(entries.length / 2));
  const avg1 = firstHalf.reduce((a, b) => a + b.price, 0) / firstHalf.length;
  const avg2 = secondHalf.reduce((a, b) => a + b.price, 0) / secondHalf.length;
  if (avg2 < avg1 * 0.97) return 'down';
  if (avg2 > avg1 * 1.03) return 'up';
  return 'stable';
}

function renderPriceTrends() {
  const history = loadPriceHistory();
  const container = document.getElementById('priceTrends');
  const entries = Object.entries(history).filter(([_, v]) => v.length > 0);

  if (entries.length === 0) {
    container.innerHTML = `
      <div style="text-align:center;padding:30px;color:var(--text-tertiary);">
        <div style="font-size:40px;margin-bottom:8px;">📊</div>
        <p>Noch keine Preiseintragungen.<br>Füge Produkte mit Preis hinzu (Modal → Preis-Feld).</p>
      </div>`;
    return;
  }

  container.innerHTML = entries.map(([name, data]) => {
    const trend = getPriceTrend(name);
    const avg = getAvgPrice(name);
    const latest = data[0]?.price || 0;
    const bars = data.slice(0, 10).reverse().map(d => {
      const height = Math.max(3, (d.price / (latest * 1.3 || 1)) * 28);
      return `<div class="pt-bar" style="height:${height}px;" title="${d.price} € (${d.date})"></div>`;
    }).join('');

    const trendIcon = trend === 'up' ? '↗' : trend === 'down' ? '↘' : '→';
    const trendClass = trend;

    return `
      <div class="price-trend">
        <span class="pt-name" style="cursor:pointer;" onclick="renderPriceHistoryFor('${name.replace(/'/g, "\'")}')">${name}</span>
        <div class="pt-chart">${bars}</div>
        <span class="pt-price">${latest.toFixed(2)} €</span>
        <span class="pt-trend ${trendClass}">${trendIcon}</span>
      </div>`;
  }).join('');
}

function renderPriceHistoryFor(productName) {
  const history = loadPriceHistory();
  const container = document.getElementById('priceHistory');
  const name = productName || '';

  if (name) {
    // Single product detail
    const entries = history[name] || [];
    if (entries.length === 0) {
      container.innerHTML = '<div style="text-align:center;padding:20px;color:var(--text-tertiary);">Keine Einträge</div>';
      return;
    }
    container.innerHTML = `
      <table class="price-history-table">
        <thead><tr><th>Datum</th><th>Preis</th></tr></thead>
        <tbody>${entries.map(e => `<tr><td>${e.date}</td><td>${e.price.toFixed(2)} €</td></tr>`).join('')}</tbody>
      </table>`;
  } else {
    // All products - show average prices
    const sorted = Object.entries(history)
      .filter(([_, v]) => v.length > 0)
      .map(([name, data]) => ({ name, entries: data, avg: getAvgPrice(name), latest: data[0]?.price }))
      .sort((a, b) => (b.avg || 0) - (a.avg || 0));

    if (sorted.length === 0) {
      container.innerHTML = '<div style="text-align:center;padding:20px;color:var(--text-tertiary);">Keine Preiseintragungen</div>';
      return;
    }

    container.innerHTML = `
      <table class="price-history-table">
        <thead><tr><th>Produkt</th><th>Durchschn.</th><th>Letzter</th><th>Einträge</th></tr></thead>
        <tbody>${sorted.map(s => `
          <tr>
            <td style="cursor:pointer;font-weight:500;" onclick="renderPriceHistoryFor('${s.name.replace(/'/g, "\'")}')">${s.name}</td>
            <td>${s.avg?.toFixed(2)} €</td>
            <td>${s.latest?.toFixed(2)} €</td>
            <td>${s.entries.length}</td>
          </tr>`).join('')}</tbody>
      </table>`;
  }
}

// ═══════════════════════════════════════════════════════════
//  SUB-SECTION NAVIGATION
// ═══════════════════════════════════════════════════════════
function showSubSection(sectionId) {
  // Hide all sections
  document.querySelectorAll('.section').forEach(s => s.classList.remove('active'));
  // Show target
  const section = document.getElementById(sectionId);
  if (section) section.classList.add('active');

  // Update sub-tabs
  document.querySelectorAll('.sub-tab').forEach(t => t.classList.remove('active'));
  const tab = document.querySelector(`.sub-tab[data-section="${sectionId}"]`);
  if (tab) tab.classList.add('active');

  // Show back button for sub-sections
  const backBtn = document.getElementById('backBtn');
  if (backBtn) backBtn.style.display = 'flex';
}
